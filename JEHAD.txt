def module_one_connections(target_ip=None):
    print("\n\033[1;36m[Module 1: Global Active Connections Monitor]\033[0m")
    if not target_ip:
        target_ip = input("Enter target IP (or press Enter to list ALL active system connections): ").strip()
    
    try:
        if target_ip:
            print(f"[*] Extracting all active sessions and socket states for target ({target_ip})...")
            cmd = f"ss -tunp | grep {target_ip}"
        else:
            print("[*] Extracting ALL active system network connections globally...")
            cmd = "ss -tunp"
            
        result = subprocess.run(cmd, shell=True, capture_output=True, text=True)
        if result.stdout.strip():
            print("\033[1;32m[+] Active Connection(s) & Sockets Found:\033[0m")
            print(result.stdout)
        else:
            print("[-] No active connections found matching the criteria.")
    except Exception as e:
        print(f"[-] Error executing connection monitor: {e}")

def module_two_audit(target_ip=None):
    print("\n\033[1;36m[Module 2: Deep Nmap Port & Service Audit (All Ports)]\033[0m")
    if not target_ip:
        target_ip = input("Enter the target IP address to audit: ").strip()
        
    print(f"[*] Running deep, comprehensive Nmap scan across ALL 65535 ports on {target_ip}...")
    try:
        os.system(f"nmap -sV -sC -p- -T4 {target_ip}")
    except Exception as e:
        print(f"[-] Error running deep nmap scan: {e}")

def module_three_vulnerability_audit(target_ip=None):
    print("\n\033[1;36m[Module 3: Full-Spectrum Vulnerability & Port Mapping Assessment]\033[0m")
    if not target_ip:
        target_ip = input("Enter target IP address for global vulnerability auditing: ").strip()
        
    print(f"[*] Performing deep socket analysis and vulnerability mapping on {target_ip}...\n")
    open_count = 0
    
    for port, info in PORT_VULN_DATABASE.items():
        try:
            s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            s.settimeout(1.5)
            result = s.connect_ex((target_ip, port))
            
            if result == 0:
                open_count += 1
                print(f"\033[1;32m[+] PORT {port} ({info['service']}) -> OPEN\033[0m")
                print(f"\033[1;31m    [!] Associated Vulnerability: {info['vuln']}\033[0m")
                print("-" * 60)
            s.close()
        except Exception:
            pass
            
    print(f"\n[+] Socket Vulnerability Mapping Complete: Detected {open_count} critical vectors.")
    print("[*] Tip: For custom ports not in the static map, Module 2 performs a full 65535-port deep audit.")

def module_four_isolation(target_ip=None):
    print("\n\033[1;36m[Module 4: Global Incident Response & Network Isolation (UFW)]\033[0m")
    if not target_ip:
        target_ip = input("Enter the target IP address to isolate immediately: ").strip()
        
    confirm = input(f"[!] WARNING: Are you sure you want to completely isolate {target_ip} globally? (y/n): ").strip().lower()
    
    if confirm == 'y':
        print(f"[*] Enforcing firewall restrictions to quarantine {target_ip}...")
        try:
            os.system(f"sudo ufw deny from {target_ip} to any")
            os.system(f"sudo ufw deny from any to {target_ip}")
            print(f"\033[1;32m[+] Target {target_ip} successfully quarantined and isolated from the network!\033[0m")
        except Exception as e:
            print(f"[-] Error applying UFW rule: {e}")
    else:
        print("[-] Isolation procedure aborted.")

def main():
    parser = argparse.ArgumentParser(description="JEHAD Global Security Auditing & Incident Response Tool")
    parser.add_argument("-t", "--target", help="Target IP address for automated scanning")
    parser.add_argument("-m", "--module", type=int, choices=[1, 2, 3, 4], help="Module number to execute (1 to 4)")
    args = parser.parse_args()

    if args.module:
        print_banner()
        if args.module == 1:
            module_one_connections(args.target)
        elif args.module == 2:
            module_two_audit(args.target)
        elif args.module == 3:
            module_three_vulnerability_audit(args.target)
        elif args.module == 4:
            module_four_isolation(args.target)
        sys.exit(0)

    while True:
        print_banner()
        print("1. Global Active Connections Monitor (All/Target Sockets)")
        print("2. Deep Nmap Port & Service Audit (All 65535 Ports)")
        print("3. Full-Spectrum Vulnerability & Port Mapping Assessment")
        print("4. Global Incident Response & Network Isolation (UFW)")
        print("5. Exit Tool")
        
        choice = input("\nEnter your choice (1-5): ").strip()
        
        if choice == '1':
            module_one_connections()
        elif choice == '2':
            module_two_audit()
        elif choice == '3':
            module_three_vulnerability_audit()
        elif choice == '4':
            module_four_isolation()
        elif choice == '5':
            print("\n[*] Exiting JEHAD Global Tool. Stay secure!")
            sys.exit(0)
        else:
            print("\n[-] Invalid choice! Please select between 1 and 5.")
        
        input("\nPress Enter to return to the main menu...")

if __name__ == "__main__":
    main()
